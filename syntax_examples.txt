## Fill triangle perspective-correct with vertex colors (vertex buffer).
## verts points to 24 consecutive int64 == 3 vertices x 8 fields each
## (x, y, z, u, v, r, g, b), so the buffer is 192 bytes (24 * 8). Texture
## color is multiplied by the interpolated per-vertex color. The texture is
## 32-bit BGRA (4 bytes/pixel), NOT the RGB565 used by fill_triangle_affine.
## Stage the 24 fields in a local array and loop the pokes -- one poke per
## element at offset i*8 -- instead of 24 hand-written mem.poke64 lines.

    x0, y0, z0, u0, v0, r0, g0, b0,
    x1, y1, z1, u1, v1, r1, g1, b1,
    x2, y2, z2, u2, v2, r2, g2, b2
};
local int verts = mem.alloc(192);
local int i = 0;
while (i < v.len) {
    mem.poke64(verts, i * 8, v[i]);
    i = i + 1;
}
fill_triangle_pcolor(verts, tex_ptr, tex_w, tex_h);

    mem.pokef32(cbuf, ci * 4, bg[ci]);
    ci = ci + 1;
}
gpu.clear(cbuf);

## Blend mode: 0 = opaque (default), 1 = straight alpha (uses per-vertex a).
gpu.set_blend(1);

## fill_triangle_gpu(verts, count, tex_ptr, tex_w, tex_h)
## verts: count*3 vertices, each 16 float32 (64-byte stride), built DIRECTLY
## with mem.pokef32 (no int64->float convert): x,y,z, u,v, r,g,b,a, slice, flag,
## nx,ny,nz, pad, pad.
##   - positions are WORLD-space (transformed by set_viewproj in the shader)
##   - u,v are raw texel coords; r,g,b,a are 0-255 -- the shader normalizes both
##   - slice: Texture2DArray layer (512x512 BGRA); NEGATIVE slice = SDF text mode
##   - flag: 1 disables per-vertex fog, 0 = full fog; NEGATIVE flag = billboard
##     point mode (camera-facing expansion)
##   - nx,ny,nz: vertex normal; the last 2 floats are padding to round to 64B
## tex_w packs the layer count in its high 32 bits: texw | (nlayers << 32).
## Stage one triangle's 48 floats (3 verts x 16) from an array + loop:
local float[] tri = {
    x0, y0, z0, u0, v0, r0, g0, b0, a0, 0.0, 0.0, nx0, ny0, nz0, 0.0, 0.0,
    x1, y1, z1, u1, v1, r1, g1, b1, a1, 0.0, 0.0, nx1, ny1, nz1, 0.0, 0.0,
    x2, y2, z2, u2, v2, r2, g2, b2, a2, 0.0, 0.0, nx2, ny2, nz2, 0.0, 0.0
};
local int gverts = mem.alloc(192);   // 3 verts * 64-byte stride
local int gi = 0;
while (gi < tri.len) {
    mem.pokef32(gverts, gi * 4, tri[gi]);
    gi = gi + 1;
}
fill_triangle_gpu(gverts, 1, tex_ptr, tex_w, tex_h);
